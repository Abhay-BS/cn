 strcpy(w, t);
 for (i = 0; w[i + 16]; i++)
 if (w[i] == '1')
 for (j = 0; j < 17; j++)
 w[i + j] ^= g[j] & 1; /* XOR */
}

#include <stdio.h>
int main() {
 int cap, rate, in[1000] = {0}, n = 0, ch, left = 0, x, sent, drop, i;
 printf("\n enter bucket size : ");
 scanf("%d", &cap);
 printf("\n enter output rate :");
 scanf("%d", &rate);
 do {
 printf("\n enter number of packets coming at second %d :", n + 1);
 scanf("%d", &in[n++]);
 printf("\n enter 1 to contiue or 0 to quit......... ");
 scanf("%d", &ch);
 } while (ch && n < 50);
 printf("\n second \t recieved \t sent \t dropped \t remained \n");
 for (i = 0; left || i < n; i++) {
 x = in[i] + left; /* packets in bucket */
 sent = x < rate ? x : rate; /* leak at most 'rate' */
 left = x - sent; /* still waiting */
 drop = left > cap ? left - cap : 0; /* overflow is dropped */
 left -= drop;
 printf("%d \t%d\t \t %d\t \t %d \t %d \n", i + 1, in[i], sent, drop, left);
 }
 return 0;
}
